// Nisa Phase 6 Firestore security rules.
// Deploy from the Firebase Console or Firebase CLI.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    // Registration and profile editing write these user-owned documents.
    // These matches were previously missing, causing the registration batch
    // to fail with PERMISSION_DENIED.
    match /users/{uid} {
      allow read, create, update, delete: if isOwner(uid);
    }

    match /public_profiles/{uid} {
      allow read: if signedIn();
      allow create, update, delete: if isOwner(uid);
    }

    match /private_users/{uid} {
      allow read, create, update, delete: if isOwner(uid);
    }

    match /preferences/{uid} {
      allow read, create, update, delete: if isOwner(uid);
    }

    // Interests: users may create their own outgoing interest.
    // Reads/updates must later be tightened to sender/receiver rules
    // once the Interest documents are migrated to Firestore.
    match /interests/{interestId} {
      allow read: if signedIn() &&
        (resource.data.senderId == request.auth.uid ||
         resource.data.receiverId == request.auth.uid);
      allow create: if signedIn() &&
        request.resource.data.senderId == request.auth.uid;
      allow update: if signedIn() &&
        (resource.data.senderId == request.auth.uid ||
         resource.data.receiverId == request.auth.uid);
      allow delete: if signedIn() &&
        resource.data.senderId == request.auth.uid;
    }

    match /matches/{matchId} {
      allow read: if signedIn() &&
        (resource.data.userA == request.auth.uid ||
         resource.data.userB == request.auth.uid);
      allow create: if signedIn() &&
        (request.resource.data.userA == request.auth.uid ||
         request.resource.data.userB == request.auth.uid);
      allow update, delete: if false;
    }

    match /conversations/{conversationId} {
      allow read, create: if signedIn() &&
        (request.auth.uid in resource.data.participants ||
         request.auth.uid in request.resource.data.participants);
      allow update, delete: if signedIn() &&
        request.auth.uid in resource.data.participants;

      match /messages/{messageId} {
        allow read, create: if signedIn() &&
          request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants;
        allow update: if signedIn() && (
          request.auth.uid == resource.data.senderId ||
          (
            request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants &&
            request.auth.uid != resource.data.senderId &&
            request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly(['deliveredAt', 'seenAt'])
          )
        );
        allow delete: if false;
      }
    }

    match /family_members/{memberId} {
      allow read, create, update, delete: if signedIn() &&
        request.auth.uid == resource.data.userId;
    }
  }
}
